cmake_minimum_required(VERSION 3.20)
project(RoClickNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(engine_tests tests/EngineTests.cpp)
target_include_directories(engine_tests PRIVATE core)
add_test(NAME engine COMMAND engine_tests)
if(MSVC)
  target_compile_options(engine_tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(engine_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
if(WIN32)
  add_library(roclick_updates windows/UpdateService.cpp)
  target_include_directories(roclick_updates PUBLIC windows)
  target_compile_definitions(roclick_updates PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(roclick_updates PUBLIC shell32 ole32 advapi32 uuid)
  add_library(roclick_engine windows/Runtime.cpp)
  target_include_directories(roclick_engine PUBLIC windows core)
  target_compile_definitions(roclick_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(roclick_engine PUBLIC user32 advapi32)
  add_executable(window_tests tests/WindowTests.cpp)
  # The fixture exercises the same executable-name filter as the installed app.
  set_target_properties(window_tests PROPERTIES OUTPUT_NAME RobloxPlayerBeta)
  target_link_libraries(window_tests PRIVATE roclick_engine)
  add_test(NAME windows_discovery COMMAND window_tests)
  set_tests_properties(windows_discovery PROPERTIES TIMEOUT 30)
  if(MSVC)
    target_compile_options(roclick_engine PRIVATE /W4 /utf-8)
  endif()
  add_executable(roclick_benchmark tests/Benchmark.cpp)
  target_compile_definitions(roclick_benchmark PRIVATE ROCLICK_BENCHMARK_GAME_MODE)
  target_link_libraries(roclick_benchmark PRIVATE roclick_engine gdi32)
  if(MSVC)
    target_compile_options(roclick_benchmark PRIVATE /W4 /utf-8)
  endif()
endif()
